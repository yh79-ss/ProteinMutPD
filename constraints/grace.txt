# Pinned dependency set for ProteinMutPD: the base install plus the `train` and `structure` extras.
#
# Use it on a login node, inside the virtual environment (scripts/setup_env.sh does this for you):
#   python -m pip install -c constraints/grace.txt '.[train]'
# A constraints file installs nothing by itself; it only fixes the versions pip may choose.
#
# "Tested" below: on 2026-11-09 the suite was run on CPU, Linux x86_64, with exactly this set, on Python 3.11.14, 3.12.3 (Ubuntu's
# own build) and 3.13.12: the CI steps (pytest -m "not slow and not gpu and not rehearsal", then pytest -m rehearsal).
# Nothing has run on Grace or on a GPU. torch comes from PyPI with CUDA 12.8 libraries (`pip install --dry-run` resolves it
# without an extra index URL; not tried on Grace's drivers). The CPU runs used the `2.10.0+cpu` wheel of the same release,
# which `==2.10.0` also matches. Python 3.14 was only resolved.

torch==2.10.0            # tested: 3.11, 3.12, 3.13 (CPU wheel); the reference environment has the cu128 wheel
transformers==4.51.0     # tested: 3.11, 3.12, 3.13
peft==0.13.2             # tested: 3.11, 3.12, 3.13
# pulled in by transformers and peft; pinned so that a reinstall gives the tested versions
tokenizers==0.21.4       # tested: 3.11, 3.12, 3.13
huggingface-hub==0.36.2  # tested: 3.11, 3.12, 3.13
safetensors==0.7.0       # tested: 3.11, 3.12, 3.13
accelerate==1.12.0       # tested: 3.11, 3.12, 3.13

# numpy 1.26.4 has no Python 3.13 wheel; 2.1.x is the first line that does. Neither numpy nor scipy is pinned for 3.14:
# the pinned versions have no 3.14 wheel.
numpy==1.26.4 ; python_version < "3.13"      # tested: 3.11, 3.12
numpy==2.1.3 ; python_version == "3.13"      # tested: 3.13
scipy==1.14.1 ; python_version < "3.14"      # tested: 3.11, 3.12, 3.13
pyyaml==6.0.3            # tested: 3.11, 3.12, 3.13

biotite==1.6.0           # tested: 3.11, 3.12, 3.13
